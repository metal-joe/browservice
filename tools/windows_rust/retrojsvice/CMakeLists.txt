# Builds the C++ vice plugin retrojsvice.dll on Windows for the Rust port
# (used by tools/build_windows_rust.ps1). Same target definition as the
# retrojsvice part of winbuild/CMakeLists.txt, as a standalone project
# because the Rust build does not need the CEF CMake project.
#
# Dependencies (Poco with NetSSL, OpenSSL, libjpeg-turbo, zlib) come from
# vcpkg through its MSBuild integration (Directory.Build.props/.targets in
# the build directory, written by the build script as tools/build_windows.py
# does), so a Visual Studio generator is required.
#
#   RETROJSVICE_DIR           path of viceplugins/retrojsvice
#   RETROJSVICE_GEN_HTML_CPP  html.cpp generated by gen_html_cpp.py

cmake_minimum_required(VERSION 3.21)
project(retrojsvice CXX)

if(NOT CMAKE_GENERATOR MATCHES "^Visual Studio")
  message(FATAL_ERROR "retrojsvice needs a Visual Studio generator (vcpkg MSBuild integration)")
endif()
if(NOT EXISTS "${RETROJSVICE_DIR}/src/html.hpp")
  message(FATAL_ERROR "RETROJSVICE_DIR='${RETROJSVICE_DIR}' is not viceplugins/retrojsvice")
endif()
if(NOT EXISTS "${RETROJSVICE_GEN_HTML_CPP}")
  message(FATAL_ERROR "RETROJSVICE_GEN_HTML_CPP='${RETROJSVICE_GEN_HTML_CPP}' does not exist")
endif()

file(GLOB_RECURSE RETROJSVICE_SRCS "${RETROJSVICE_DIR}/src/*.cpp" "${RETROJSVICE_DIR}/src/*.hpp")
add_library(retrojsvice SHARED ${RETROJSVICE_SRCS} "${RETROJSVICE_GEN_HTML_CPP}")
# The generated file includes "../src/html.hpp"; outside of
# viceplugins/retrojsvice/gen it is found relative to this include directory.
target_include_directories(retrojsvice PRIVATE "${RETROJSVICE_DIR}/src")
target_compile_features(retrojsvice PRIVATE cxx_std_17)
